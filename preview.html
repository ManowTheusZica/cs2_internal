// Features/InventoryChanger.h

struct InventoryItem {
  int weaponDefIndex;    // Ex: 7 = AK-47, 507 = Karambit
  int paintKit;          // Paint kit ID
  int seed;              // Pattern seed
  float wear;           // Float (0.0 = Factory New)
  int stattrak;         // StatTrak kills
  bool isKnife;         // Se e faca
  bool isGlove;         // Se e luva
};

class InventoryChanger {
  void AddItem(const InventoryItem& item);
  void Tick();            // Aplica skins a cada arma
  void SetupDefaults();  // AK Redline, AWP Dragon Lore...
};

// Exemplo de uso:
InventoryChanger ic;
ic.AddItem({ 507, 38, 100, 0.001f, -1, 6, "Karambit | Fade", true, false });
ic.Tick(); // Aplica a cada frame
